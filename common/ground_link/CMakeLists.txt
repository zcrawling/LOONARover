if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  cmake_minimum_required(VERSION 3.22)
  project(loonar_ground_link_tools LANGUAGES C CXX)
  include(CTest)
  set(CMAKE_C_STANDARD 99)
  set(CMAKE_C_STANDARD_REQUIRED ON)
  set(CMAKE_C_EXTENSIONS OFF)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  add_library(loonar_project_options INTERFACE)
  add_library(loonar_project_warnings INTERFACE)
  target_compile_options(loonar_project_warnings INTERFACE
    -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()

add_library(loonar_ground_link STATIC src/frame.cpp)
target_include_directories(loonar_ground_link PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(loonar_ground_link PUBLIC loonar_project_options loonar_project_warnings)

add_executable(ground_link_mock src/mock.cpp)
target_link_libraries(ground_link_mock PRIVATE loonar_ground_link)

add_library(loonar_ground_wire_c STATIC
  ../../cfs/apps/common/loonar_ground_wire.c)
target_include_directories(loonar_ground_wire_c PUBLIC ../../cfs/apps/common)
target_link_libraries(loonar_ground_wire_c PUBLIC loonar_project_options loonar_project_warnings)

if(BUILD_TESTING)
  add_executable(test_ground_link_frame tests/test_frame.cpp)
  target_link_libraries(test_ground_link_frame PRIVATE loonar_ground_link)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(test_ground_link_frame PRIVATE -UNDEBUG)
  endif()
  add_test(NAME ground_link.frame COMMAND test_ground_link_frame)
  add_executable(test_ground_link_wire_c tests/test_wire.c)
  target_link_libraries(test_ground_link_wire_c PRIVATE loonar_ground_wire_c)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(test_ground_link_wire_c PRIVATE -UNDEBUG)
  endif()
  add_test(NAME ground_link.wire_c COMMAND test_ground_link_wire_c)
endif()
