cmake_minimum_required(VERSION 3.16)
project(loonar_terrain_core)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(Eigen3 REQUIRED)
  ament_add_gtest(test_geometry test/test_geometry.cpp)
  target_include_directories(test_geometry PRIVATE include)
  target_link_libraries(test_geometry Eigen3::Eigen)
  target_compile_features(test_geometry PRIVATE cxx_std_17)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_algorithms test/test_algorithms.py)
endif()
ament_package()
