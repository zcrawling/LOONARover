cmake_minimum_required(VERSION 3.16)
project(loonar_terrain_perception)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp rclcpp_action sensor_msgs geometry_msgs tf2_ros tf2_eigen loonar_msgs loonar_terrain_core grid_map_core grid_map_ros)
  find_package(${dep} REQUIRED)
endforeach()
find_package(Eigen3 REQUIRED)
add_executable(terrain_perception src/terrain_perception.cpp)
target_compile_features(terrain_perception PRIVATE cxx_std_17)
target_compile_options(terrain_perception PRIVATE -Wall -Wextra)
target_link_libraries(terrain_perception Eigen3::Eigen)
ament_target_dependencies(terrain_perception rclcpp rclcpp_action sensor_msgs geometry_msgs tf2_ros tf2_eigen loonar_msgs loonar_terrain_core grid_map_core grid_map_ros)
install(TARGETS terrain_perception DESTINATION lib/${PROJECT_NAME})
ament_package()
