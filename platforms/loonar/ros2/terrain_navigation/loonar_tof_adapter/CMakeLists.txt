cmake_minimum_required(VERSION 3.16)
project(loonar_tof_adapter)
find_package(ament_cmake REQUIRED)
set(CUBEEYE_SDK "" CACHE PATH "CubeEye SDK release directory (optional)")
if(CUBEEYE_SDK)
  add_executable(cubeeye_capture src/cubeeye_capture.cpp)
  target_compile_features(cubeeye_capture PRIVATE cxx_std_17)
  target_include_directories(cubeeye_capture PRIVATE ${CUBEEYE_SDK}/include/CubeEye)
  target_link_directories(cubeeye_capture PRIVATE ${CUBEEYE_SDK}/lib)
  target_link_libraries(cubeeye_capture CubeEye pthread)
  # CubeEye's own DT_NEEDED libraries live beside the SDK, not in ld.so.conf.
  # DT_RUNPATH only resolves direct dependencies; DT_RPATH also resolves the
  # vendor's OpenCV/FFmpeg/live555 dependencies without a shell LD_LIBRARY_PATH.
  target_link_options(cubeeye_capture PRIVATE "-Wl,--disable-new-dtags")
  file(GLOB vendor_library_dirs LIST_DIRECTORIES true "${CUBEEYE_SDK}/thirdparty/*/lib")
  list(FILTER vendor_library_dirs EXCLUDE REGEX "/python/lib$")
  list(PREPEND vendor_library_dirs "${CUBEEYE_SDK}/lib")
  list(APPEND vendor_library_dirs "${CUBEEYE_SDK}/thirdparty/liblive555/lib/Release")
  foreach(library_dir IN LISTS vendor_library_dirs)
    target_link_options(cubeeye_capture PRIVATE "-Wl,-rpath-link,${library_dir}")
  endforeach()
  set_target_properties(cubeeye_capture PROPERTIES INSTALL_RPATH "${vendor_library_dirs}")
  install(TARGETS cubeeye_capture DESTINATION lib/${PROJECT_NAME})
endif()
install(DIRECTORY scripts/ DESTINATION lib/${PROJECT_NAME} USE_SOURCE_PERMISSIONS)
ament_package()
